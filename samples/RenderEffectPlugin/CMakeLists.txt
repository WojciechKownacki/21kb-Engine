cmake_minimum_required(VERSION 3.24)
project(RenderEffectPlugin LANGUAGES CXX)
if(NOT KB_RENDER_EFFECT_INCLUDE_DIR)
    message(FATAL_ERROR "Set KB_RENDER_EFFECT_INCLUDE_DIR to the SDK renderer include directory")
endif()
add_library(sample_render_effect MODULE GradientEffect.cpp)
target_compile_features(sample_render_effect PRIVATE cxx_std_20)
target_include_directories(sample_render_effect PRIVATE "${KB_RENDER_EFFECT_INCLUDE_DIR}")
set_target_properties(sample_render_effect PROPERTIES PREFIX "")
